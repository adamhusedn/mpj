// ─── Proxy pool ──────────────────────────────────────────────────────────────
let PROXIES = (config.proxies || [])
    .map(l => String(l).trim())
    .filter(l => l && !l.startsWith('#'))
    .map(l => l.includes('://') ? l : `http://${l}`);

    console.log(chalk.green(`[Proxy] Loaded ${PROXIES.length} proxies dari configmpj.json`));

let proxyIndex = 0;
function getNextProxy() {
    if (!PROXIES.length) return null;
    const proxy = PROXIES[proxyIndex % PROXIES.length];
    proxyIndex++;
    return proxy;
}

/** Buat HttpsProxyAgent dari proxy string, atau null jika tidak ada proxy */
function makeAgent() {
    const proxy = getNextProxy();
    return proxy ? new HttpsProxyAgent(proxy) : undefined;
}

// ─── CapSolver: buat task ─────────────────────────────────────────────────────
const functionGetTokenAction = (sitekey, linkAccess, version) =>
    new Promise((resolve, reject) => {
        fetch(
                `https://api.capsolver.com/createTask`, {
                    method: "post",
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        clientKey: keyCaptcha,
                        task: {
                            type: "ReCaptchaV2TaskProxyLess",
                            websiteURL: linkAccess,
                            websiteKey: sitekey
                        }
                    })
                }
            )
            .then((res) => res.json())
            .then((data) => {
                if (data.errorId === 0 && data.taskId) resolve(data.taskId);
                else reject(new Error(data.errorDescription || data.errorCode || 'Failed to submit captcha'));
            })
            .catch((err) => reject(err));
    });

// ─── CapSolver: ambil hasil task ──────────────────────────────────────────────
const functionGetRealTokenAction = (taskId) =>
    new Promise((resolve, reject) => {
        fetch(
                `https://api.capsolver.com/getTaskResult`, {
                    method: "post",
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        clientKey: keyCaptcha,
                        taskId: taskId
                    })
                }
            )
            .then((res) => res.json())
            .then((text) => {
                resolve(text);
            })
            .catch((err) => reject(err));
    });

async function solveRecaptcha(sitekey = "6LdPw1kUAAAAANnq6qTKBXXUyFzbZbjXdQkZt_4m", linkAccess = "https://www.mpj.id/en/user/login") {
    console.log(chalk.yellow('[Captcha] Submitting task to CapSolver...'));
    const taskId = await functionGetTokenAction(sitekey, linkAccess, "v2");
    console.log(chalk.yellow(`[Captcha] Task ID: ${taskId}. Polling for token...`));

    for (let attempt = 0; attempt < 25; attempt++) {
        await new Promise(r => setTimeout(r, 5000));
        const res = await functionGetRealTokenAction(taskId);
        if (res.errorId !== 0) {
            throw new Error(`CapSolver error: ${res.errorDescription || res.errorCode}`);
        }
        if (res.status === "ready") {
            console.log(chalk.green('[Captcha] ✅ Solved successfully!'));
            return res.solution.gRecaptchaResponse;
        }
        // status "processing" — masih diproses, tunggu lagi
        process.stdout.write(chalk.gray('.'));
    }
    throw new Error('CapSolver timed out (>125s)');
}

async function checkCaptchaBalance() {
    try {
        const res = await fetch(`https://api.capsolver.com/getBalance`, {
            method: 'post',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ clientKey: keyCaptcha })
        });
        const data = await res.json();
        if (data.errorId === 0) {
            console.log(chalk.green(`[CapSolver] Saldo CapSolver: $${data.balance}`));
            return parseFloat(data.balance);
        } else {
            console.log(chalk.red(`[CapSolver] Gagal cek saldo: ${data.errorDescription || JSON.stringify(data)}`));
        }
    } catch (err) {
        console.log(chalk.red(`[CapSolver] Error cek saldo: ${err.message}`));
    }
    return null;
}

                       '0831','0832','0833','0838','0895','0896','0897','0898','0899'];
    const mobile_no = pick(prefixes) + String(rnd(10000000, 99999999));

    // Tanggal lahir: 1980–2000, usia min 18 thn
    const year      = rnd(1980, 2000);
    const month     = String(rnd(1, 12)).padStart(2, '0');
    const day       = String(rnd(1, 28)).padStart(2, '0');
    const birthdate = `${year}${month}${day}`;

    // Gender random
    const gender    = Math.random() < 0.5 ? '01' : '02';

    // PIN 6 digit
    const pin       = String(rnd(100000, 999999));

    // Password acak tapi kuat

    return { name, mobile_no, birthdate, gender, pin, password, pref_cinema, fingerprint };
}

    const ua    = randomUA();

        .split(';')
        .map(c => c.trim())
        .filter(c => !c.startsWith('token='))
        .join('; ');
    const authCookie = `${cleanCookie}; token=${token}`;
    return fetch('https://www.mpj.id/api/coupon/register', {
        method: 'POST',
        agent,
        headers: {
            'Host': 'www.mpj.id',
            'X-Csrf-Token': csrf,
            'Authorization': `Bearer ${token}`,
            'Accept-Language': 'id-ID,id;q=0.9',
            'Content-Type': 'application/json',
            'User-Agent': ua,
            'Accept': '*/*',
            'Origin': 'https://www.mpj.id',
            'Sec-Fetch-Site': 'same-origin',
            'Sec-Fetch-Mode': 'cors',
            'Sec-Fetch-Dest': 'empty',
            'Referer': 'https://www.mpj.id/en/membership/my_coupon',

function randomIndonesianUsername() {
    const depan    = NAMA_DEPAN[Math.floor(Math.random() * NAMA_DEPAN.length)];
    const belakang = NAMA_BELAKANG[Math.floor(Math.random() * NAMA_BELAKANG.length)];
    const angka    = Math.floor(Math.random() * 9000) + 100;
    const sep      = ['', '.', '_'][Math.floor(Math.random() * 3)];
    return `${depan}${sep}${belakang}${angka}`;
}


/** Fetch available domains from jestco.id (cached) */
let _jestcoDomains = [];
async function getJestcoDomains() {
    if (_jestcoDomains.length) return _jestcoDomains;
    const res  = await fetch(`${JESTCO_API}?action=get_public_config&_t=${Date.now()}`, {
        headers: JESTCO_HEADERS
    });
    const data = await res.json();
    _jestcoDomains = data.domains || [];
    return _jestcoDomains;
}

/**
 * Generate a temp email address using jestco.id.
 * No account creation needed — inbox accessible via email address directly.
 * @returns {{ email: string }}
 */
async function createJestcoEmail() {
    const domains = await getJestcoDomains();
    if (!domains.length) throw new Error('[jestco] No domains available');
    const domain   = domains[Math.floor(Math.random() * domains.length)];
    const username = randomIndonesianUsername();
    const email    = `${username}@${domain}`;
    console.log(chalk.cyan(`[jestco] Temp email: ${email}`));
    return { email };
}

/**
 * Poll jestco.id inbox until a new message arrives.
 * @param {string} email    - temp email address
 * @param {number} retries  - max attempts
 * @param {number} delayMs  - ms between attempts
 * @returns {{ from, subject, body } | null}
 */
async function waitForJestcoMessage(email, retries = 20, delayMs = 5000) {
    for (let i = 0; i < retries; i++) {
        console.log(chalk.yellow(`[Inbox] Checking inbox... attempt ${i + 1}/${retries}`));

        const res = await fetch(
            `${JESTCO_API}?to=${encodeURIComponent(email)}&site_pass=&_t=${Date.now()}`,
            { headers: JESTCO_HEADERS }
        );
        const messages = await res.json();

        if (Array.isArray(messages) && messages.length > 0) {
            const first   = messages[0];
            const from    = first.from || first.sender || '';
            const subject = first.subject || '';
            const body    = first.body || first.text || first.html || first.content || JSON.stringify(first);
            console.log(chalk.green(`[Inbox] Message received! From: ${from} | Subject: ${subject}`));
            return { from, subject, body };
        }

        if (i < retries - 1) {
            await new Promise(r => setTimeout(r, delayMs));
        }
    }
    return null;
}

/**
 * Extract OTP code (4-8 digit number) from email body text.
 */
function readOtpFromEmail(body) {
    const patterns = [
        /\b(\d{6})\b/,   // Most common: 6 digits
        /\b(\d{4})\b/,   // 4 digits
        /\b(\d{8})\b/,   // 8 digits
        /OTP[:\s]+(\d{4,8})/i,
        /code[:\s]+(\d{4,8})/i,
        /kode[:\s]+(\d{4,8})/i,
        /verif\w*[:\s]+(\d{4,8})/i,
    ];

    for (const pattern of patterns) {
        const match = body.match(pattern);
        if (match) {
            const digits = (match[1] || match[0]).replace(/\D/g, '');
            if (digits.length >= 4 && digits.length <= 8) {
                return digits;
            }
        }
    }
    return null;
}

(async () => {
    await checkCaptchaBalance();

    ngocok: while(true) {
        try {
            // 1. Buat temp email baru via jestco.id (tanpa perlu daftar akun)
            const { email } = await createJestcoEmail();

            // 2. Get mpj cookie & CSRF
            const gettingCookie = await getCookie();
            if (gettingCookie.data.includes('This may occur when unusual or automated activity is detected.')) {
                console.log(chalk.red('[mpj] Rate limited, tunggu 30s...'));
                await new Promise(r => setTimeout(r, 30000));
                continue ngocok;
            }
            const cleanedCookies = gettingCookie.headers.map(cookie => cookie.split(';')[0]);
            let cookie = cleanedCookies.join('; ');

            // Validasi CSRF tersedia — jika tidak ada, proxy gagal atau mpj block
            if (!cookie.includes('csrf-token=')) {
                console.log(chalk.red('[mpj] CSRF token tidak ditemukan (proxy gagal/block). Retry...'));
                await new Promise(r => setTimeout(r, 5000));
                continue ngocok;
            }
            let csrf = cookie.split('csrf-token=')[1].split(';')[0];


            console.log(chalk.blue(`[mpj] Sending OTP to: ${email}`));

            // 3. Kirim OTP ke temp email via mpj
            const sendingEmail = await sendEmail(cookie, csrf, email);
            console.log(chalk.green(`[mpj] OTP send response:`), JSON.stringify(sendingEmail.data));

            if (!sendingEmail.data || sendingEmail.data.status !== true) {
                const isRateLimit = sendingEmail.data?.status_code === 429;
                console.log(chalk.red(`[mpj] Failed: ${sendingEmail.data?.message || 'unknown error'}. ${isRateLimit ? 'Rate limited, tunggu 30s...' : 'Retrying...'}` ));
                await new Promise(r => setTimeout(r, isRateLimit ? 30000 : 2000));
                continue ngocok;
            }

            // 4. Tunggu email OTP masuk ke jestco.id (max ~100 detik)
            console.log(chalk.yellow('[Inbox] Waiting for OTP email...'));
            const msg = await waitForJestcoMessage(email, 20, 5000);

            if (!msg) {
                console.log(chalk.red('[Inbox] No email received within timeout. Retrying...'));
                continue ngocok;
            }

            // 5. Ekstrak OTP dari body email
            const fullText = `${msg.subject} ${msg.body}`;
            const otp = readOtpFromEmail(fullText);

            if (!otp) {
                console.log(chalk.red('[OTP] Could not extract OTP code.'));
                console.log(chalk.gray('[OTP] Email body:'), fullText.slice(0, 300));
                continue ngocok;
            }

            console.log(chalk.green(`[OTP] ✅ OTP Code: ${otp}`));

            // 6. Verify OTP ke mpj
            console.log(chalk.blue('[mpj] Verifying OTP...'));
            const verifyRes = await verifyOtp(cookie, csrf, email, otp);
            console.log(chalk.green('[mpj] Verify response:'), JSON.stringify(verifyRes));

            if (!verifyRes || verifyRes.status !== true) {
                console.log(chalk.red(`[mpj] OTP verify failed: ${verifyRes?.message || 'unknown'}. Retrying...`));
                await new Promise(r => setTimeout(r, 2000));
                continue ngocok;
            }

            // 7. Register akun mpj dengan data random
            const userData = randomUserData(email);
            console.log(chalk.blue(`[mpj] Registering: ${userData.name} | ${userData.mobile_no} | ${userData.birthdate}`));
            const regRes = await registerAccount(cookie, csrf, email, otp, userData);
            console.log(chalk.green('[mpj] Register response:'), JSON.stringify(regRes));

            if (regRes && regRes.status === true) {
                const mbr = regRes.MBR_CRD_NO || '-';

                // 8. Login setelah register berhasil (solve captcha dulu via CapSolver)
                let loginToken = '-';
                let loginOk = false;
                try {
                    console.log(chalk.blue('[mpj] Solving captcha for login via CapSolver...'));
                    const captchaResponse = await solveRecaptcha("6LdPw1kUAAAAANnq6qTKBXXUyFzbZbjXdQkZt_4m", "https://www.mpj.id/en/user/login");
                    console.log(chalk.blue('[mpj] Logging in with captcha token...'));
                    const loginRes = await loginmpj(cookie, csrf, email, "dowercorp@123", captchaResponse);
                    console.log(chalk.green('[mpj] Login response:'), JSON.stringify(loginRes.data));

                    const sessionCookie = (loginRes.cookies || [])
                        .map(c => c.split(';')[0])
                        .find(c => c.includes('mpj_session') || c.includes('PHPSESSID') || c.includes('session'));

                    loginToken = loginRes.data?.token || loginRes.data?.access_token || sessionCookie || '-';
                    loginOk    = loginRes.data?.status === true || !!loginRes.data?.token || loginRes.data?.status_code === 200;

                    // Update cookie dengan cookies baru dari response login
                    if (loginRes.cookies && loginRes.cookies.length > 0) {
                        const newCookies = loginRes.cookies.map(c => c.split(';')[0]);
                        const merged = new Map();
                        cookie.split(';').forEach(c => {
                            const [k, ...v] = c.trim().split('=');
                            if (k) merged.set(k, v.join('='));
                        });
                        newCookies.forEach(c => {
                            const [k, ...v] = c.trim().split('=');
                            if (k) merged.set(k, v.join('='));
                        });
                        cookie = Array.from(merged.entries()).map(([k, v]) => `${k}=${v}`).join('; ');
                        if (merged.has('csrf-token')) csrf = merged.get('csrf-token');
                    }

                    // 9. Register Coupon jika login berhasil dan memiliki token
                    if (loginOk && loginRes.data?.token) {
                        code: while(true) {

                            console.log(chalk.blue(`[mpj] Registering coupon: ${couponCode}...`));
                            const couponRes = await registerCoupon(cookie, csrf, loginRes.data.token, couponCode);

                        }
                    }
                } catch (captchaErr) {
                    console.log(chalk.red(`[mpj] Captcha / Login error: ${captchaErr.message}`));
                }

                const line = `${email}|dowercorp@123|${userData.name}|${userData.mobile_no}|${mbr}|${loginToken}`;

                console.log(chalk.red(`[mpj] Register failed: ${regRes?.message || 'unknown'}. Retrying...`));

        }

function generateCode() {
  const letters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  const numbers = '0123456789';

  const randomLetter = () =>
    letters[Math.floor(Math.random() * letters.length)];

  const randomNumber = () =>
    numbers[Math.floor(Math.random() * numbers.length)];

  return `mpjMOV${randomLetter()}${randomLetter()}${randomLetter()}${randomNumber()}${randomNumber()}COLA`;
}

function getTimestamp() {
    const now = new Date();
    const pad = (n) => String(n).padStart(2, '0');
    const hours = pad(now.getHours());
    const minutes = pad(now.getMinutes());
    const seconds = pad(now.getSeconds());
    return `[${hours}:${minutes}:${seconds}]`;
}
